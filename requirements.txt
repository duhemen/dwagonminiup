# 9. INSTALLATION CHECKLIST
# ============================================================

Prerequisites:
  [ ] Node.js 20+ installed       (node --version)
  [ ] pnpm 8+ installed           (pnpm --version)
  [ ] Docker Desktop running      (docker ps)
  [ ] Git installed               (git --version)

Setup:
  [ ] Clone repository
  [ ] Run: pnpm install
  [ ] Run: pnpm approve-builds    (pilih 'a' → Enter)
  [ ] Copy .env.example → .env
  [ ] Run: docker-compose up -d postgres redis
  [ ] Run: npx prisma db push --accept-data-loss
  [ ] Run: npx prisma generate
  [ ] Run: npx tsx prisma/seed.ts

Run Services:
  [ ] Terminal 1: pnpm --filter central-server dev
  [ ] Terminal 2: .\start-all-edges.ps1
  [ ] Terminal 3: pnpm --filter dashboard-ui dev

Verify:
  [ ] http://localhost:4000/api/health    → {"status":"ok"}
  [ ] http://localhost:4001/api/health    → region: sumatera
  [ ] http://localhost:4007/api/health    → region: papua
  [ ] http://localhost:5173               → Login page muncul
  [ ] Login: emen@dwagon.id / demo123     → Berhasil masuk

# 11. USEFUL COMMANDS
# ============================================================

  # Development
  pnpm install                                 Install deps
  pnpm --filter central-server dev             Start central
  pnpm --filter edge-node dev                  Start 1 edge
  .\start-all-edges.ps1                        Start 7 edges
  pnpm --filter dashboard-ui dev               Start frontend

  # Database
  npx prisma db push                           Sync schema
  npx prisma generate                          Regen client
  npx prisma studio                            GUI at :5555
  npx tsx prisma/seed.ts                       Seed data

  # Docker
  docker-compose up -d postgres redis          Start DB
  docker-compose down                          Stop all
  docker ps                                    Check status

  # Production
  .\deploy.ps1 init                            Init prod env
  .\deploy.ps1 build                           Build images
  .\deploy.ps1 up                              Start all
  .\deploy.ps1 backup                          Backup DB
  .\deploy.ps1 ssl                             Request SSL

  # Troubleshoot
  Get-Process node | Stop-Process -Force       Kill all node
  docker logs dwagon-postgres                  View PG logs
  docker exec -it dwagon-postgres psql -U dwagon -d dwagon    DB shell
